\title{A survey of the lantern court\\[3pt]{\large Notes from the winter trial}}
\author{Survey desk\thanks{Drawings use a common scale.} \and Model workshop}
\date{}\maketitle
\begin{abstract}Three paper screens were placed along a model courtyard. We recorded the first position at which the rear lantern disappeared.\end{abstract}
\section{Arrangement}A narrow screen stood beside each gate. The observer moved one tile at a time.\footnote{A tile measures one model unit.}
\section{Result}The middle gate concealed the lantern first. Its screen crossed the sight line at tile four.
