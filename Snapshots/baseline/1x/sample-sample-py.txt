#!/usr/bin/env python3
"""Counts the words of a text file, the most frequent first."""
import sys
from collections import Counter
def count_words(text: str) -> Counter:
words = (word.strip(".,!?\"'()").lower() for word in text.split())
return Counter(word for word in words if word)
def main(path: str, limit: int = 10) -> None:
with open(path, encoding="utf-8") as file:
counts = count_words(file.read())
for word, count in counts.most_common(limit):
print(f"{count:5d} {word}") # 数を右にそろえる
if __name__ == "__main__":
main(sys.argv[1])
