// A small booklet planner: how many sheets a saddle-stitched booklet needs.
import Foundation
struct Booklet {
let pages: Int
/// Pages are padded to a multiple of four, one sheet for every four.
var sheets: Int { (pages + 3) / 4 }
func side(of sheet: Int, front: Bool) -> (left: Int, right: Int) {
let total = sheets * 4
return front
? (total - 2 * sheet, 2 * sheet + 1)
: (2 * sheet + 2, total - 2 * sheet - 1)
}
}
let booklet = Booklet(pages: 10)
for sheet in 0..<booklet.sheets {
let front = booklet.side(of: sheet, front: true)
let back = booklet.side(of: sheet, front: false)
print("用紙 \(sheet + 1): 表 [\(front.left)|\(front.right)] 裏 [\(back.left)|\
(back.right)]")
}
